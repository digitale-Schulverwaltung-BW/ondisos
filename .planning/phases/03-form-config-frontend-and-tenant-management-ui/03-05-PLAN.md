New FormConfig DB-backed pattern (from RESEARCH.md Pattern 1):
```php
private static array $cache = [];  // keyed by formKey

public static function get(string $formKey): ?array
{
    if (array_key_exists($formKey, self::$cache)) {
        return self::$cache[$formKey];
    }
    $db = Database::getConnection();
    $tenantId = TenantContext::getTenantId();
    $stmt = $db->prepare(
        'SELECT config_json FROM form_configs WHERE form_key = ? AND tenant_id = ? LIMIT 1'
    );
    $stmt->bind_param('si', $formKey, $tenantId);
    $stmt->execute();
    $row = $stmt->get_result()->fetch_assoc();
    $config = $row ? json_decode($row['config_json'], true) : null;
    self::$cache[$formKey] = $config;
    return $config;
}

// Required for test isolation (Pitfall 5 from RESEARCH.md):
public static function reset(): void { self::$cache = []; }
```

    ```php
    // Validate TenantContext was resolved (slug must be valid)
    try {
        $tenantId = TenantContext::getTenantId();
    } catch (\RuntimeException $e) {
        http_response_code(401);
        echo json_encode(['error' => 'Unauthorized: invalid or missing tenant']);
        exit;

    if ($formKey === '') {
        http_response_code(400);
        echo json_encode(['error' => 'Missing form parameter']);
        exit;
    }

    $config = FormConfig::get($formKey);
    if ($config === null) {
        http_response_code(404);
        echo json_encode(['error' => 'Form not found']);
        exit;
    }

    echo json_encode(['success' => true, 'config' => $config], JSON_UNESCAPED_UNICODE);
    ```
